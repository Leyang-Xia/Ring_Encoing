cmake_minimum_required(VERSION 3.16)
project(ringlab VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT UNIX)
  message(FATAL_ERROR "The optimizer process adapter currently requires POSIX (Linux/macOS).")
endif()
add_library(ringlab_core src/json.cpp src/bitvector.cpp src/field.cpp src/circuit.cpp src/cse.cpp)
target_include_directories(ringlab_core PUBLIC include)
add_library(ringlab_experiments src/matrices.cpp src/verification.cpp src/search.cpp src/backend.cpp)
target_link_libraries(ringlab_experiments PUBLIC ringlab_core)
add_executable(ringlab apps/ringlab.cpp)
target_link_libraries(ringlab PRIVATE ringlab_experiments)
target_compile_definitions(ringlab PRIVATE RINGLAB_ROOT="${CMAKE_CURRENT_SOURCE_DIR}")
file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/third_party/xsets/X-Sets.cpp" XSETS_SHA256)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${CMAKE_CURRENT_SOURCE_DIR}/third_party/xsets/X-Sets.cpp")
target_compile_definitions(ringlab_experiments PRIVATE RINGLAB_XSETS_SHA256="${XSETS_SHA256}")
foreach(target ringlab_core ringlab_experiments ringlab)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
add_executable(xsets third_party/xsets/X-Sets.cpp)
add_executable(uber third_party/uber/Uber.cpp)
include(CTest)
if(BUILD_TESTING)
  add_executable(core-test tests/core_test.cpp)
  target_link_libraries(core-test PRIVATE ringlab_core)
  add_test(NAME core COMMAND core-test "${CMAKE_CURRENT_SOURCE_DIR}/data/certificates")
  add_test(NAME certificates COMMAND ringlab verify --exhaustive)
  add_test(NAME cli COMMAND bash "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_test.sh" $<TARGET_FILE:ringlab>)
  set_tests_properties(cli PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(BUILD_TESTING)
  add_executable(experiment-test tests/experiment_test.cpp)
  target_link_libraries(experiment-test PRIVATE ringlab_experiments)
  add_test(NAME experiment-regressions COMMAND experiment-test "${CMAKE_CURRENT_SOURCE_DIR}/data/certificates")
endif()
